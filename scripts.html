document.addEventListener('DOMContentLoaded', function () { const registrationForm = document.getElementById('registrationForm'); if (registrationForm) { registrationForm.addEventListener('submit', function (e) { e.preventDefault(); const name = document.getElementById('name').value.trim(); const email = document.getElementById('email').value.trim(); const password = document.getElementById('password').value.trim(); const user = { name, email, password }; let users = JSON.parse(localStorage.getItem('users')) || []; users.push(user); localStorage.setItem('users', JSON.stringify(users)); alert('User registered successfully!'); registrationForm.reset(); }); } });
